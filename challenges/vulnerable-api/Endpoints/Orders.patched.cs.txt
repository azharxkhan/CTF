// The fix: use a method that PARAMETERIZES the interpolation. FromSqlRaw treats the string
// as literal SQL; FromSql / FromSqlInterpolated turn {r} into a DbParameter.

// --- Option A: FromSql (EF Core 7+) — note it's FromSql, NOT FromSqlRaw ---
var orders = db.Orders
    .FromSql($"SELECT * FROM Orders WHERE Ref = {r}")   // {r} becomes a parameter
    .ToList();

// --- Option B: FromSqlInterpolated (same effect, explicit name) ---
var orders = db.Orders
    .FromSqlInterpolated($"SELECT * FROM Orders WHERE Ref = {r}")
    .ToList();

// --- Option C: plain LINQ — parameterized automatically, no raw SQL at all ---
var orders = db.Orders.Where(o => o.Ref == r).ToList();
