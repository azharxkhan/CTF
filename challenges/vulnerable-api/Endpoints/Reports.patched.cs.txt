// CHALLENGE 7 — SECURE reference (reviewer's answer key). NOT compiled into the app.
// The fix: resolve the full path and confirm it stays INSIDE the reports directory before
// reading. Reject anything that escapes the root.

app.MapGet("/api/reports/download", (string file) =>
{
    var root = Path.GetFullPath(ReportsDir(env));

    // Resolve the requested path to its real, absolute form...
    var requested = Path.GetFullPath(Path.Combine(root, file));

    // ...then require it to live under the reports root. Path separator guard stops the
    // "/reports-evil" prefix trick.
    if (!requested.StartsWith(root + Path.DirectorySeparatorChar, StringComparison.Ordinal))
        return Results.BadRequest("Invalid file name.");

    if (!File.Exists(requested)) return Results.NotFound();
    return Results.File(File.ReadAllBytes(requested), "application/octet-stream",
        Path.GetFileName(requested));
});
